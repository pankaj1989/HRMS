## Report metadata schema

For each report:

```typescript
{
  reportCode: 'headcount-by-department',
  reportName: 'Headcount by Department',
  category: 'headcount-and-organization',
  description: 'Active employee count broken down by department',
  
  visibleToRoles: ['hr-manager', 'hr-head', 'tenant-admin', 'department-head'],
  scopeOptions: ['all-tenant', 'entity', 'department'],
  defaultScope: 'entity',
  
  availableFilters: [
    {
      filterCode: 'as-of-date',
      label: 'As of Date',
      field: 'asOfDate',
      type: 'date-picker',
      defaultValue: 'today',
      isRequired: true,
    },
    {
      filterCode: 'employment-type',
      label: 'Employment Type',
      field: 'employmentType',
      type: 'multi-select',
      options: [...],
      isRequired: false,
    },
    // ...
  ],
  
  dimensions: [
    { dimensionCode: 'department', label: 'Department', field: 'departmentId', type: 'categorical' },
    { dimensionCode: 'category', label: 'Category', field: 'category', type: 'categorical' },
  ],
  
  metrics: [
    { metricCode: 'headcount', label: 'Headcount', aggregation: 'count', field: '_id' },
    { metricCode: 'avg-tenure-months', label: 'Avg Tenure (months)', aggregation: 'avg', field: 'tenureMonths' },
  ],
  
  defaultVisualization: 'bar-chart',
  availableVisualizations: ['bar-chart', 'table', 'pie-chart'],
  
  useReadModel: true,
  readModelCode: 'daily-headcount-snapshot',
  
  cacheTtlSeconds: 300,
  exportFormats: ['csv', 'excel', 'pdf'],
}
```
